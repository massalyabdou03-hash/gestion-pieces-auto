async function loadPiecesSelect() { const { data, error } = await supabaseClient .from("pieces").select("id, designation, reference_interne, quantite_stock").eq("actif", true).order("designation"); if (error) return; if (!data) return; const sel = document.getElementById("piece_id"); data.forEach(p => { piecesMap[p.id] = p; const opt = document.createElement("option"); opt.value = p.id; opt.textContent = `${p.designation} (${p.reference_interne}) — stock: ${p.quantite_stock}`; sel.appendChild(opt); }); }
